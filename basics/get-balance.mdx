description: Get SOL balance
---

# New Way
```tsx
import { address, createSolanaRpc, decimalFixedPointToString, lamportsToSol } from "@solana/kit";

const rpc = createSolanaRpc("https://api.devnet.solana.com");

const balance = await rpc.getBalance(address('MTSLZDJppGh6xUcnrSSbSQE5fgbvCtQ496MqgQTv8c1')).send();
console.log(decimalFixedPointToString(lamportsToSol(balance.value)));
```

# Old Way

```tsx
import { Connection, LAMPORTS_PER_SOL, PublicKey } from "@solana/web3.js";

const connection = new Connection("https://api.devnet.solana.com");

const publicKey = new PublicKey("MTSLZDJppGh6xUcnrSSbSQE5fgbvCtQ496MqgQTv8c1");

const getBalance = async () => {
    let balance = await connection.getBalance(publicKey);
    console.log(`${balance / LAMPORTS_PER_SOL} SOL`);
};

getBalance();
```
