---
title: Get Transaction
description: Fetch a confirmed transaction
---

```js
import { createSolanaRpc, mainnet, signature } from "@solana/kit";

const rpc = createSolanaRpc(mainnet("https://api.mainnet-beta.solana.com"));
const transaction = await rpc.getTransaction(
    signature('56K4Cdwk9YKpSTZruMNbkhuykj2E55HtCnN8o2j8xUdsthj9G8qPiMwVQBxEXWJ3yht2WGXbxjPpPCqjiSLWAGAA'),
    { encoding: 'json', maxSupportedTransactionVersion: 0 },
).send();
console.log(transaction);
```
